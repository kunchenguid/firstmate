class LockBusy(RuntimeError):
    pass


def _sha256(path: Path) -> str | None:
    if not path.is_file():
        return None
    return hashlib.sha256(path.read_bytes()).hexdigest()


def _source_commit(root: Path) -> str:
    try:
        completed = subprocess.run(
            ["git", "-C", str(root), "rev-parse", "HEAD"],
            capture_output=True,
            text=True,
            check=True,
            timeout=10,
        )
    except (OSError, subprocess.SubprocessError):
        return "unknown"
    return completed.stdout.strip() or "unknown"


def _source_snapshot(paths: list[Path]) -> dict[str, Any]:
    entries = []
    for path in paths:
        digest = _sha256(path)
        entries.append({"path": str(path), "sha256": digest, "exists": digest is not None})
    return {"files": entries, "snapshot_hash": hashlib.sha256(json.dumps(entries, sort_keys=True).encode()).hexdigest()}


def _acquire(path: Path):
    path.parent.mkdir(parents=True, exist_ok=True)
    handle = path.open("a+", encoding="utf-8")
    try:
        fcntl.flock(handle.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
    except OSError as exc:
        handle.close()
        raise LockBusy(str(path)) from exc
    return handle


def run_pipeline(
    *,
    source_paths: list[Path],
    output: Path,
    lock_path: Path = DEFAULT_LOCK,
    runner: Runner,
    held_lock: Path | None = None,
) -> dict[str, Any]:
    test_handle = None
    try:
        if held_lock is not None:
            test_handle = _acquire(held_lock)
        lock_handle = _acquire(lock_path)
    except Exception:
        if test_handle is not None:
            fcntl.flock(test_handle.fileno(), fcntl.LOCK_UN)
            test_handle.close()
        raise
    try:
        context: dict[str, Any] = {
            "run_id": datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ"),
            "source_commit": _source_commit(ROOT),
            "snapshot": _source_snapshot(source_paths),
        }
        output.parent.mkdir(parents=True, exist_ok=True)
        snapshot_artifact = output.parent / f"{output.stem}.snapshot.json"
        snapshot_artifact.write_text(
            json.dumps(context["snapshot"], ensure_ascii=False, indent=2, sort_keys=True) + "\n",
            encoding="utf-8",
        )
        results: list[dict[str, Any]] = []
        report: dict[str, Any] = {
            "schema_version": 1,
            "status": "running",
            "run_id": context["run_id"],
            "source_commit": context["source_commit"],
            "phase_order": list(PHASES),
            "organize_runs": 0,
            "snapshot": context["snapshot"],
            "snapshot_artifact": str(snapshot_artifact),
            "sections": {
                "ssot": "docs_ssot",
                "memory": "organize",
                "skills": "skill_review",
            },
            "phases": results,
        }
        for phase in PHASES:
            if phase == "organize":
                report["organize_runs"] += 1
            try:
                result = runner(phase, context)
                if not isinstance(result, dict):
                    result = {"ok": True, "value": result}
                result = {"phase": phase, "source_commit": context["source_commit"], **result}
                results.append(result)
                if result.get("ok") is False:
                    raise RuntimeError(str(result.get("error", "phase failed")))
            except Exception as exc:
                report.update({"status": "failed", "failed_phase": phase, "error": str(exc)})
                output.parent.mkdir(parents=True, exist_ok=True)
                output.write_text(json.dumps(report, ensure_ascii=False, indent=2, sort_keys=True) + "\n", encoding="utf-8")
                return report
        report["status"] = "success"
        output.parent.mkdir(parents=True, exist_ok=True)
        output.write_text(json.dumps(report, ensure_ascii=False, indent=2, sort_keys=True) + "\n", encoding="utf-8")
        return report
    finally:
        fcntl.flock(lock_handle.fileno(), fcntl.LOCK_UN)
        lock_handle.close()
        if test_handle is not None:
            fcntl.flock(test_handle.fileno(), fcntl.LOCK_UN)
            test_handle.close()


def _fixture_runner(fail_phase: str | None) -> Runner:
    def run(phase: str, context: dict[str, Any]) -> dict[str, Any]:
        if phase == fail_phase:
            raise RuntimeError(f"fixture failure: {phase}")
        return {"ok": True, "source_snapshot_hash": context["snapshot"]["snapshot_hash"]}

    return run


def _coverage_commands(root: Path) -> list[tuple[str, list[str]]]:
    python = sys.executable
    return [
        ("artifact", [python, str(root / "AutomationSync" / "artifact_index.py"), "--check"]),
        ("pointers", [python, str(root / "AutomationSync" / "artifact_index.py"), "--check-pointers"]),
        ("manifest", [python, str(root / "AutomationSync" / "skills_registry.py"), "--check-manifest"]),
        ("runtime", [python, str(root / "AutomationSync" / "skills_registry.py"), "--check-runtime"]),
        ("mirrors", [python, str(root / "automation-core" / "scripts" / "check_mirrors.py")]),
        ("drift", [python, str(root / "AutomationSync" / "runtime_instruction_drift_auditor.py"), "--json"]),
        ("instruction-surface", [python, str(root / "AutomationSync" / "instruction_surface_check.py")]),
    ]


def _run_coverage(root: Path, report_dir: Path) -> dict[str, Any]:
    report_dir.mkdir(parents=True, exist_ok=True)
    checks = []
    for name, command in _coverage_commands(root):
        artifact = report_dir / f"coverage-{name}.log"
        try:
            completed = subprocess.run(command, capture_output=True, text=True, cwd=str(root), timeout=120)
            output = completed.stdout + completed.stderr
            returncode = completed.returncode
        except Exception as exc:
            output = f"coverage command failed before completion: {type(exc).__name__}: {exc}\n"
            returncode = 1
        artifact.write_text(output, encoding="utf-8")
        checks.append({"name": name, "ok": returncode == 0, "returncode": returncode, "artifact": str(artifact)})
    failed = [check for check in checks if not check["ok"]]
    return {"ok": not failed, "checks": checks, "failed": [check["name"] for check in failed]}


def _production_runner(report_dir: Path, ledger: Path) -> Runner:
    python = sys.executable
    env = os.environ.copy()
    env["PYTHONPATH"] = str(ROOT / "automation-core") + os.pathsep + env.get("PYTHONPATH", "")
    env["CODE_REVIEW_FORCE"] = "1"

    def run(phase: str, context: dict[str, Any]) -> dict[str, Any]:
        current_commit = context.get("source_commit", "unknown")
        if phase == "snapshot":
            report_dir.mkdir(parents=True, exist_ok=True)
            (report_dir / "snapshot.json").write_text(
                json.dumps(context["snapshot"], ensure_ascii=False, indent=2, sort_keys=True) + "\n",
                encoding="utf-8",
            )
            return {
                "ok": True,
                "artifact": str(report_dir / "snapshot.json"),
                "source_snapshot_hash": context["snapshot"]["snapshot_hash"],
                "source_commit": current_commit,
            }
        if phase == "coverage":
            result = _run_coverage(ROOT, report_dir)
            result["source_snapshot_hash"] = context["snapshot"]["snapshot_hash"]
            result["source_commit"] = current_commit
            return result
        if phase == "docs_ssot":
            command = ["/bin/bash", str(ROOT / "AutomationSync" / "docs_ssot_audit_runner.sh")]
            phase_env = {**env, "SUNDAY_DOCUMENT_SNAPSHOT": context["snapshot"]["snapshot_hash"]}
        elif phase == "organize":
            try:
                from retro_doctrine_autolink import sync_retro_stow_candidates
                sync_retro_stow_candidates(root_dir=ROOT)
            except Exception:
                pass
            if not ledger.is_file():
                return {
                    "ok": True,
                    "skipped": "missing-ledger",
                    "source_snapshot_hash": context["snapshot"]["snapshot_hash"],
                    "source_commit": current_commit,
                }
            command = [python, str(ROOT / "AutomationSync" / "memory_governance.py"), "--organize-live", "--input", str(ledger), "--output", str(report_dir / "organized.json")]
            phase_env = env
        elif phase == "doctrine_review":
            try:
                from retro_doctrine_autolink import sync_retro_stow_candidates
                sync_retro_stow_candidates(root_dir=ROOT)
            except Exception:
                pass
            command = [str(ROOT / "local-bin" / "code-review-runner.sh"), "layer2", (ROOT / "AutomationSync" / "prompts" / "doctrine-review.txt").read_text(encoding="utf-8"), "doctrine-review"]
            phase_env = {**env, "CODE_REVIEW_CODEX_MODEL": "gpt-5.6-sol"}
        elif phase == "tool_incidents":
            command = [python, str(ROOT / "AutomationSync" / "tool_incident_weekly_report.py"), "--no-send"]
            phase_env = env
        elif phase == "claude_md_doctor":
            command = [python, str(ROOT / "AutomationSync" / "claude_md_doctor_weekly.py")]
            phase_env = env
        else:
            command = [python, str(ROOT / "AutomationSync" / "skills_review.py")]
            phase_env = {**env, "SUNDAY_COORDINATOR": "1"}
        report_dir.mkdir(parents=True, exist_ok=True)
        timeout = 900 if phase in ("docs_ssot", "doctrine_review", "skill_review") else 300
        completed = subprocess.run(command, capture_output=True, text=True, env=phase_env, cwd=str(ROOT), timeout=timeout)
        artifact = report_dir / f"{phase}.log"
        snapshot_line = (
            f"source_commit={current_commit}\n"
            f"phase={phase}\n"
            f"source_snapshot_hash={context['snapshot']['snapshot_hash']}\n"
        )
        artifact.write_text(snapshot_line + completed.stdout + completed.stderr, encoding="utf-8")
        return {
            "ok": completed.returncode == 0,
            "returncode": completed.returncode,
            "artifact": str(artifact),
            "source_snapshot_hash": context["snapshot"]["snapshot_hash"],
            "source_commit": current_commit,
        }

    return run


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Run Sunday document governance as one ordered job")
    parser.add_argument("--fixture", type=Path)
    parser.add_argument("--fail-phase", choices=PHASES)
    parser.add_argument("--output", type=Path, default=ROOT / "AutomationSync" / "runtime" / "logs" / "sunday-document-governance.json")
    parser.add_argument("--lock-path", type=Path, default=DEFAULT_LOCK)
    parser.add_argument("--ledger", type=Path, default=DEFAULT_LEDGER)
    args = parser.parse_args(argv)
    report_dir = args.output.parent / "sunday-document-governance"
    if args.fixture:
        sources = [args.fixture]
        runner = _fixture_runner(args.fail_phase)
    else:
        sources = _production_source_paths(ROOT, args.ledger)
        runner = _production_runner(report_dir, args.ledger)
    try:
        report = run_pipeline(source_paths=sources, output=args.output, lock_path=args.lock_path, runner=runner)
    except LockBusy as exc:
        print(json.dumps({"status": "busy", "error": str(exc)}, ensure_ascii=False))
        return 75
    print(json.dumps(report, ensure_ascii=False, indent=2, sort_keys=True))
    return 0 if report["status"] == "success" else 1


if __name__ == "__main__":
    raise SystemExit(main())
